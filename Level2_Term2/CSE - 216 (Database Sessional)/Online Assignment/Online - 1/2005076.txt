1.
SELECT (FIRST_NAME || ' ' || LAST_NAME) FULLNAME, SALARY
FROM EMPLOYEES
WHERE  VSIZE(FIRST_NAME) + VSIZE(LAST_NAME) > 10
ORDER BY SALARY ASC;

2.
SELECT TO_CHAR(HIRE_DATE,'MONTH') MONTH_NAME, COUNT(*) TOTAL_EMPLOYEES
FROM EMPLOYEES
GROUP BY TO_CHAR(HIRE_DATE,'MONTH'), TO_CHAR(HIRE_DATE,'MM')
ORDER BY TO_CHAR(HIRE_DATE,'MM');

3.
SELECT MANAGER_ID, COUNT(*) AS TOTAL_EMPLOYEES , ROUND(AVG(SALARY),2) AVG_SAL
FROM EMPLOYEES
GROUP BY MANAGER_ID
HAVING COUNT(*) > 4 AND AVG(SALARY) > 3000
ORDER BY TOTAL_EMPLOYEES DESC, AVG_SAL ASC;

4.
SELECT DEPARTMENT_NAME, COUNT(E.LAST_NAME) TOTAL_EMPLOYEES, ROUND(AVG(E.SALARY),2 ) AVG_SAL
FROM DEPARTMENTS D JOIN EMPLOYEES E
ON D.DEPARTMENT_ID = E.DEPARTMENT_ID
GROUP BY DEPARTMENT_NAME
ORDER BY TOTAL_EMPLOYEES DESC;

5.
SELECT (SUBSTR(FIRST_NAME,1,1) || '. ' || LAST_NAME) FULLNAME, (12*SALARY + 12*SALARY* (NVL(COMMISSION_PCT,0))) YEARLY_INCOME
FROM EMPLOYEES
WHERE SUBSTR(LAST_NAME,1,1) IN ('M','m')
ORDER BY YEARLY_INCOME DESC;
