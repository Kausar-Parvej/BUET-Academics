--1
SELECT TITLE
FROM COURSE C1
WHERE EXISTS
(
SELECT *
FROM TAKES T1
WHERE C1.COURSE_ID = T1.COURSE_ID
AND C1.DEPT_NAME = 'Comp. Sci.' 
OR C1.DEPT_NAME = 'Math' 
)
OR NOT EXISTS
(
SELECT *
FROM TAKES T1
WHERE C1.COURSE_ID = T1.COURSE_ID
AND C1.DEPT_NAME = 'Elec. Eng.' 
)
ORDER BY TITLE;

--2
SELECT  DISTINCT S.NAME
FROM STUDENT S JOIN ADVISOR A
ON S.ID = A.S_ID
JOIN TEACHES T ON A.I_ID = T.ID
AND EXISTS
(
SELECT T2.COURSE_ID
FROM TAKES T2
WHERE T2.COURSE_ID = T.COURSE_ID
)
ORDER BY S.NAME;
--3
SELECT 
(
SELECT S1.NAME
FROM STUDENT S1
WHERE S1.ID=T1.ID
) NAME
FROM TAKES T1 
WHERE T1.COURSE_ID =
(
SELECT COURSE_ID
FROM COURSE C1
WHERE C1.COURSE_ID = T1.COURSE_ID
AND C1.TITLE = 'Image Processing'
);

--4
SELECT I1.ID, I1.NAME
FROM INSTRUCTOR I1
WHERE 5<= 
(
SELECT COUNT(*)
FROM TEACHES T
WHERE I1."ID" = T."ID"
GROUP BY T.ID
)
ORDER BY I1."NAME";

--5
SELECT 
(SELECT I1."NAME" 
FROM INSTRUCTOR I1
WHERE I1.ID=T1.ID
) INSTRUCTOR_NMAE,

(SELECT C.TITLE
FROM COURSE C
WHERE C.COURSE_ID=T1.COURSE_ID
) COURSE_TITLE,
T1.YEAR

FROM TEACHES T1
WHERE EXISTS
(
SELECT *
FROM TEACHES T2
WHERE T1.COURSE_ID = T2.COURSE_ID
AND T1.YEAR = T2.YEAR - 1
);
