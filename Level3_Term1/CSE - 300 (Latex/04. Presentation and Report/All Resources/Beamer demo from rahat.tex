  \begin{frame}{Time Complexity analysis of deletion in AVL Tree}
    \begin{algorithm}
    \caption{Delete operation in AVL tree}
    \begin{algorithmic}[1]
    \Function{Delete}{$root, key$}
        \If{$root$ is null}
            \State \Return $root$                      \Comment{O(1)}
        \EndIf
        \If{$key < root.val$}
            \State $root.left \gets \textsc{Delete}(root.left, key)$   \Comment{O(log n) on average}
        \ElsIf{$key > root.val$}
            \State $root.right \gets \textsc{Delete}(root.right, key)$ \Comment{O(log n) on average}
        \Else
            \If{$root$ has no left child}
                \State $temp \gets root.right$
                \State $root \gets$ null
                \State \Return $temp$                              \Comment{O(1)}
            \ElsIf{$root$ has no right child}
                \State $temp \gets root.left$
                \State $root \gets$ null
                \State \Return $temp$                              \Comment{O(1)}
            \Else
                \State $temp \gets \textsc{FindMinValueNode}(root.right)$   \Comment{O(log n) on average}
                \State $root.val \gets temp.val$
                \State $root.right \gets \textsc{Delete}(root.right, temp.val)$  \Comment{O(log n) on average}
            \EndIf
        \EndIf
        \State $root.height \gets 1 + \max(\textsc{GetHeight}(root.left), \textsc{GetHeight}(root.right))$ \Comment{O(1)}
        \State $balance \gets \textsc{GetBalance}(root)$                    \Comment{O(1)}
        \If{$balance > 1$ and $\textsc{GetBalance}(root.left) \geq 0$}
            \State \Return $\textsc{RotateRight}(root)$                     \Comment{O(1)}
        \EndIf
        \If{$balance < -1$ and $\textsc{GetBalance}(root.right) \leq 0$}
            \State \Return $\textsc{RotateLeft}(root)$                      \Comment{O(1)}
        \EndIf
        \If{$balance > 1$ and $\textsc{GetBalance}(root.left) < 0$}
            \State $root.left \gets \textsc{RotateLeft}(root.left)$         \Comment{O(1)}
            \State \Return $\textsc{RotateRight}(root)$                     \Comment{O(1)}
        \EndIf
        \If{$balance < -1$ and $\textsc{GetBalance}(root.right) > 0$}
            \State $root.right \gets \textsc{RotateRight}(root.right)$      \Comment{O(1)}
            \State \Return $\textsc{RotateLeft}(root)$                      \Comment{O(1)}
        \EndIf
        \State \Return $root$                                              \Comment{O(1)}
    \EndFunction
    \end{algorithmic}
    \end{algorithm}
\end{frame}
