#include <avr/io.h>
#define  F_CPU 1000000
#include <util/delay.h>

#define row_ddr DDRB
#define col_ddr DDRA
#define row_data PORTB
#define col_data PORTA

char rows[] = {1,2,4,8,16,32,64,128};

char ALPHA[][8] ={
  {0x00,0x38,0x20,0x38,0x20,0x38,0x00,0x00} //E
};

char rotateALPHA[][8]={ // clockWise for E
  {0x00,0x38,0x20,0x38,0x20,0x38,0x00,0x00},
  {0x00,0x00,0x00,0xf8,0xa8,0xa8,0x00,0x00},
  {0x00,0xe0,0x20,0xe0,0x20,0xe0,0x00,0x00},
  {0x00,0xa8,0xa8,0xf8,0x00,0x00,0x00,0x00}
};

void move_show(){
  int i = 0,j=0,k;
  while(1){
    for(k=0;k<150;k++){
      row_data = rows[i];
      col_data = ~rotateALPHA[j][i];
      _delay_ms(2);
      i+=1;
      if(i>7) i = 0;
    }
    j+=1;
    if(j>3){
      j=0;
      i=0;
    }
    
  }
}

void static_show(){
  int i = 0;
  while(1){
    row_data = rows[i];
    col_data = ~ALPHA[0][i];
    _delay_ms(2);
    i+=1;
    if(i>7) i = 0;
  }
}

void flush_show(){
  while(1){
    for(int i=0;i<500;i++){
      row_data = rows[i%8];
      col_data = ~ALPHA[0][i%8];
      _delay_ms(2);
    }
    
    for(int i=0;i<500;i++){
      row_data = 0x00;
      col_data = 0xff;
      _delay_ms(2);
    }
  }
}

int main(void)
{
  row_ddr =0xff;
  col_ddr = 0xff;
  move_show();
}
